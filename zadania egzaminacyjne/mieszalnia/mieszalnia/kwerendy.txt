SELECT COUNT(id) as klienci
FROM klienci;

SELECT imie, nazwisko, zamowienia.id, kod_koloru, pojemnosc, data_odbioru
FROM klienci INNER JOIN zamowienia ON klienci.id=zamowienia.id_klienta
ORDER BY data_odbioru ASC;

SELECT imie, nazwisko, zamowienia.id, kod_koloru, pojemnosc, data_odbioru
FROM klienci INNER JOIN zamowienia ON klienci.id=zamowienia.id_klienta
WHERE data_odbioru BETWEEN '2021-11-05' AND '2021-11-07'
ORDER BY data_odbioru ASC;

SELECT imie, nazwisko 
FROM klienci
WHERE plec='k';
